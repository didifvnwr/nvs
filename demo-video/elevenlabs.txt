בְּרוּכִים הַבָּאִים לַמֶּרְכָּזִיָּה שֶׁל הַקּוֹל הֶחָדָשׁ וּמִיצִי תִּקְשׁוֹרֶת. בַּסֶּרְטוֹן הַקָּצָר הַזֶּה נַכִּיר לָכֶם אֶת כָּל מָה שֶׁהַמֶּרְכָּזִיָּה שֶׁלָּכֶם יוֹדַעַת לַעֲשׂוֹת.

הַדָּבָר הַיְּחִידִי שֶׁנִּדְרָשׁ הוּא אִינְטֶרְנֶט יַצִּיב. הַמֶּרְכָּזִיָּה פּוֹעֶלֶת בֶּעָנָן, וְלָכֵן אֶפְשָׁר לְחַבֵּר טֶלֶפוֹנִים מִכָּל מָקוֹם בָּעוֹלָם: מֵהַמִּשְׂרָד, מֵהַבַּיִת אוֹ מֵחוּץ לָאָרֶץ.

כָּל שִׂיחָה מַגִּיעָה לַמָּקוֹם הַנָּכוֹן. אֶפְשָׁר לִבְחוֹר: נַתָּב שִׂיחוֹת רָגִיל, אוֹ בּוֹט קוֹלִי, בְּקוֹל נָשִׁי אוֹ גַּבְרִי, שֶׁעוֹנֶה רִאשׁוֹן, מַקְשִׁיב לְמָה שֶׁהַלָּקוֹחַ צָרִיךְ וּמַעֲבִיר אוֹתוֹ לָאָדָם הַמַּתְאִים. אֶפְשָׁר גַּם לְנַתֵּב לְפִי מַחְלָקָה, כִּשּׁוּרִים וְעוֹמֶס. כְּשֶׁכֻּלָּם עֲסוּקִים, הַלָּקוֹחַ נִכְנָס לְתוֹר מַמְתִּינִים, וְיָכוֹל לָצֵאת מִמֶּנּוּ וּלְקַבֵּל אֶס־אֶם־אֶס אוֹ הוֹדָעָה מִבּוֹט וָואטְסְאַפּ. וּבִזְמַן הַהַמְתָּנָה אֶפְשָׁר גַּם לַעֲבוֹר לְשִׂיחָה עִם גַּבִּי, הַסּוֹכֶנֶת הַקּוֹלִית. יֵשׁ מַעֲנֶה אֶחָד לַיּוֹם וּמַעֲנֶה אַחֵר לַלַּיְלָה, וְהוֹדָעָה קוֹלִית מַגִּיעָה יָשָׁר לַמֵּייל.

כָּל שִׂיחָה מוּקֶלֶטֶת, וְהַהַקְלָטוֹת נִשְׁמָרוֹת בְּהַתְאָמָה לְצָרְכֵי הַלָּקוֹחַ. וְלָכֶל שִׂיחָה יֵשׁ גַּם תִּמְלוּל מָלֵא, בִּכְתָב.

בְּסוֹף כָּל שִׂיחָה, נִשְׁלָח לַמֵּייל שֶׁל הָעוֹבֵד אוֹ שֶׁל הַמְּנַהֵל סִכּוּם קָצָר שֶׁלָּהּ: מִי הִתְקַשֵּׁר, מָה נֶאֱמַר וּמָה צָרִיךְ לַעֲשׂוֹת. וְאֶפְשָׁר לִשְׁלוֹחַ לַלָּקוֹחַ פְּרָטִים, בְּמִסְרוֹן אוֹ בְּמֵייל, לְפִי בְּחִירָתְכֶם.

הַמֶּרְכָּזִיָּה מִתְחַבֶּרֶת לְרוֹב מַעֲרָכוֹת הַסִּי־אָר־אֶם, לְכָל מַעֲרֶכֶת שֶׁמְּאַפְשֶׁרֶת חִבּוּר בְּאֵיי־פִּי־אַיי. הַשִּׂיחוֹת וְהַפְּרָטִים מִסְתַּנְכְּרְנִים לְבַד.

אֶפְשָׁר לִשְׁלוֹחַ וּלְקַבֵּל אֶס־אֶם־אֶס מֵהַמֶּרְכָּזִיָּה, לִרְאוֹת דּוּחוֹת בִּזְמַן אֱמֶת, וְיֵשׁ גַּם בּוֹט וָואטְסְאַפּ עֲסָקִי. וְגַבִּי, סוֹכֶנֶת קוֹלִית עִם בִּינָה מְלָאכוּתִית, שֶׁעוֹנָה לַלְּקוֹחוֹת בְּכָל שָׁעָה.

רוֹצִים לְשַׁנּוֹת מַשֶּׁהוּ, לְמָשָׁל שְׁעוֹת פְּעִילוּת? אֶפְשָׁר לָתֵת הוֹרָאָה לְכָל עוֹזֵר אֵיי־אָיי שֶׁאַתֶּם רְגִילִים לַעֲבוֹד אִתּוֹ. הוּא מַכִּיר אֶת הַמַּעֲרֶכֶת, מִתְחַבֵּר אֵלֶיהָ מֵרָחוֹק וּמְסַדֵּר הַכֹּל.

וְאִם אַתֶּם מַעְדִּיפִים, שִׁלְחוּ הוֹדָעָה אוֹ פְּנִיָּה בְּוָואטְסְאַפּ, וְהַצֶּוֶות יְטַפֵּל בָּכֶם בִּמְהִירוּת.

מִיצִי תִּקְשׁוֹרֶת. מֶרְכָּזִיָּה שֶׁעוֹשָׂה הַכֹּל.
